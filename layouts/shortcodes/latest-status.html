{{- /* Lists the latest N status posts across all content/status// folders, rendered as theme cards (same look as the "Why Lottie4J" cards). Usage: {{< latest-status 3 >}} (N defaults to 3) */ -}} {{- $n := int (.Get 0 | default 3) -}} {{- $posts := where site.RegularPages "Section" "status" -}} {{- $posts = first $n $posts.ByDate.Reverse -}} {{- $cards := slice -}} {{- range $posts -}} {{- /* Prefer the front-matter description; otherwise use the post's first paragraph. Hugo's .Summary starts at the top of the content, so for posts that open with a lead-in followed by a section heading it glued that heading onto the excerpt (e.g. "... first release... Fighting Complexity ..."). Picking the first

skips headings and lists. */ -}} {{- $desc := .Description -}} {{- if not $desc -}} {{- with index (findRE `(?s)]*>.*?

` .Content 1) 0 -}} {{- $desc = . | plainify | truncate 120 -}} {{- end -}} {{- end -}} {{- if not $desc -}} {{- $desc = .Summary | plainify | truncate 120 -}} {{- end -}} {{- $cards = $cards | append (dict "title" .Title "href" .Path "icon" "newspaper" "content" (printf "

%s

%s

" (.Date.Format "Jan 2, 2006") $desc) ) -}} {{- end -}}
